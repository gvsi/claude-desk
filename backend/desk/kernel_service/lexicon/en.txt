# English words free prose may not use: amounts, percentages, dates, promises and bookings reach a
# visitor only in cards rendered from the business's records. One `<category>\t<regex>` per line,
# matched case-insensitively as whole words on the prose with accents and other marks dropped, each
# run of whitespace read as one space, every dash and hyphen as "-" ("next‐week" with U+2010 is
# "next-week") and every apostrophe as "’" ("oʼclock" is "o’clock"). Categories: number, money,
# percent and date (number_in_prose), and promise and booking (promise_word). A booking claim
# with a condition straight before it ("once you're booked") passes (`BOOKING_CONDITION` in
# reply_check.py). Both lexicons run on every draft, whatever its language.

# Spelled-out numbers. A bare "one" counts only next to a unit or another number word, so "one
# moment" passes.
number	two|three|four|five|six|seven|eight|nine|ten|eleven|twelve
number	(?:thir|four|fif|six|seven|eigh|nine)teen
number	(?:twen|thir|for|four|fif|six|seven|eigh|nine)ty
number	hundreds?|thousands?|millions?|billions?|dozens?|half|halves
number	zeros?|zeroes
# A quarter is an amount, a fraction and a time ("a quarter past"), blocked whichever it is.
number	quarters?
number	thirds?
number	one(?:[\s-]+and)?[\s-]+(?:hundreds?|thousands?|millions?|billions?|dozens?|half|dollars?|bucks?|cents?|euros?|pesos?|usd|per[\s-]?cent|percent)

# Money. "grand" is a thousand dollars ("a couple grand"), and is blocked in every sense ("Grand
# Avenue" too).
money	dollars?|bucks?|cents?|usd|eur|euros?|pesos?|mxn
money	grand

# Percentages
percent	percent(?:age)?s?|per[\s-]?cent|pct

# Days, months and times. "May" counts only capitalized after a word in the same sentence or after
# in, on, by, until, till, next, this, early, late, mid or last, so the modal "may" passes. "Same
# day" is a date too ("same-day", "same day"), but not across a sentence break ("the same. Day").
date	mondays?|tuesdays?|wednesdays?|thursdays?|fridays?|saturdays?|sundays?
date	january|february|march|april|june|july|august|september|october|november|december
date	(?:(?<=[^\W\d_] )|(?<=[^\W\d_], )|(?:in|on|by|until|till|next|this|early|late|mid|last)[\s-]+)(?-i:May)
date	today|tonight|tomorrow|yesterday|weekends?|noon|midnight|o['’]?clock
date	(?:next|this|last)[\s-]+(?:week|month|year)|this[\s-]+(?:morning|afternoon|evening)
date	[ap]\.\s?m\.?|pm
date	same[\s-]{0,3}day
# A time soon: "in" or "within" and up to three words before minutes or hours ("in an hour",
# "in a couple of hours", "within the hour").
date	(?:in|within)[\s-]+(?:[^\W\d_]+[\s-]+){0,3}(?:minutes?|mins?|hours?|hrs?)
date	right[\s-]+away
# "first thing in the morning" wherever it stands, and "first thing" as a time, with no
# determiner before it ("Maria can come first thing"), so advice passes ("the first thing to check").
date	first[\s-]+thing[\s-]+in[\s-]+the[\s-]+morning|(?<!the )(?<!a )(?<!one )(?<!your )(?<!my )(?<!our )(?<!his )(?<!her )(?<!their )first[\s-]+thing

# Promises only the owner's rules or a card can make. "free" passes only in "feel free" and in
# "free to" after "you're", "you are", "you'll be" or "you will be" ("you're free to ask"), where
# it can't be a price. Anywhere else "free to" may state one: "free to everyone", "free to all
# customers", "inspections are free to members". So does "free to" before a quantifier ("most"),
# determiner ("these"), pronoun, a word that starts an audience ("new", "first-time") or an audience,
# whatever comes before it: "as a member you are free to all classes", "you're free to new members".
# A verb after "you're free to" passes: "you're free to choose any day".
promise	guarant(?:ee|ees|eed|eeing|y)|promis(?:e|es|ed|ing)
promise	(?<!feel )free(?! to\b)
promise	(?<!feel )(?<!you['’]re )(?<!you are )(?<!you['’]ll be )(?<!you will be )free(?= to\b)
promise	free(?=[\s-]+to[\s-]+(?:all|any(?:one|body)?|every(?:one|body)?|each|both|some|most|several|many|no|unlimited)\b)
promise	free(?=[\s-]+to[\s-]+(?:the|an?|this|that|these|those|our|your|my|his|her|its|their|you|them)\b)
promise	free(?=[\s-]+to[\s-]+(?:new|existing|first[\s-]?time|current|returning|regular|local)\b)
promise	free(?=[\s-]+to[\s-]+(?:members?|customers?|clients?|guests?|patrons?|subscribers?|visitors?|locals|residents?)\b)
promise	free(?=[\s-]+to[\s-]+(?:students?|seniors?|kids|children|famil(?:y|ies)|patients?)\b)
promise	discount(?:s|ed|ing)?|refund(?:s|ed|ing|able)?|waiv(?:e|es|ed|er|ing)|complimentary
promise	no[\s-]+charge|on[\s-]+the[\s-]+house|warrant(?:y|ies)|money[\s-]+back
promise	(?:at[\s-]+)?no[\s-]+(?:extra[\s-]+|additional[\s-]+)?cost
# "on us" is a price of nothing ("it's on us"), but not after "count", "rely" or "depend"
# ("you can count on us").
promise	(?<!count )(?<!counts )(?<!counted )(?<!counting )(?<!rely )(?<!relies )(?<!relied )(?<!relying )(?<!depend )(?<!depends )(?<!depended )(?<!depending )on[\s-]+us
promise	no[\s-]+(?:[^\W\d_]+[\s-]+){0,2}fees?

# Booking confirmations, which only a booking card makes: "you're booked", "I've booked you",
# "your visit is confirmed", "your next appointment's been set", "you've been booked".
booking	you(?:['’]?re|[\s-]+are)[\s-]+(?:(?:all|now|already)[\s-]+)?(?:booked|confirmed|scheduled|set)
booking	(?:i|we)(?:['’]ve|[\s-]+have)?[\s-]+(?:booked|scheduled|confirmed)[\s-]+(?:you|your[\s-]+(?:booking|appointment|visit))
booking	(?:your|the)[\s-]+(?:[^\W\d_]+[\s-]+)?(?:booking|appointment|visit)(?:[\s-]+(?:is|has[\s-]+been)|['’]s(?:[\s-]+been)?)[\s-]+(?:(?:all|now|already)[\s-]+)?(?:booked|confirmed|scheduled|set)
booking	you(?:['’]ve|[\s-]+have)[\s-]+been[\s-]+(?:booked|confirmed|scheduled)
